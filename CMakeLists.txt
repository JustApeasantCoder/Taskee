cmake_minimum_required(VERSION 3.24)
project(Taskee LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
add_library(TaskeeTap SHARED src/tap.cpp src/tap.def)
add_library(TaskeePanel SHARED src/panel.cpp src/panel.def)
target_compile_definitions(TaskeePanel PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(TaskeePanel PRIVATE windowsapp runtimeobject ole32)
add_executable(TaskeeBridge src/bridge.cpp)
add_executable(TaskeePolicyTests src/policy_tests.cpp)
add_test(NAME taskbar-policy COMMAND TaskeePolicyTests)
if(BUILD_TESTING)
  add_executable(TaskeeHistoryTests src/history_tooltip_tests.cpp src/history_tooltip_tests.manifest)
  target_compile_definitions(TaskeeHistoryTests PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_compile_options(TaskeeHistoryTests PRIVATE /W4 /EHsc /utf-8)
  target_link_libraries(TaskeeHistoryTests PRIVATE windowsapp runtimeobject ole32 user32)
  add_test(NAME history-tooltip COMMAND TaskeeHistoryTests)
  set_tests_properties(history-tooltip PROPERTIES TIMEOUT 15)
endif()
target_compile_definitions(TaskeeBridge PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(TaskeeBridge PRIVATE ole32 user32)
set_property(TARGET TaskeePanel TaskeeBridge PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(TaskeeLoadTest src/load.cpp)
target_compile_definitions(TaskeeLoadTest PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(TaskeeLoadTest PRIVATE ole32 user32 shell32 propsys)
target_compile_definitions(TaskeeTap PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(TaskeeTap PRIVATE windowsapp runtimeobject ole32)
add_executable(TaskeeTest src/test.cpp)
target_compile_definitions(TaskeeTest PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(TaskeeTest PRIVATE ole32 user32 shell32 propsys)
foreach(target TaskeeTap TaskeeTest TaskeePanel TaskeeBridge TaskeeLoadTest TaskeePolicyTests)
  target_compile_options(${target} PRIVATE /W4 /EHsc /utf-8)
endforeach()
